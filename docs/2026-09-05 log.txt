```
PS C:\Users\Administrator> Get-GPOReport -Name "Default Domain Policy" -ReportType Xml

```
PS C:\Users\Administrator> Get-GPOReport -Name "Default Domain Policy" -ReportType Html -Path "$env:TEMP\Default-Domain-Policy.html"
PS C:\Users\Administrator> Invoke-Item "$env:TEMP\Default-Domain-Policy.html"
```
